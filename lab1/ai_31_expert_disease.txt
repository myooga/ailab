:- dynamic known/2.

ask(Symptom) :-
    known(Symptom, yes), !.

ask(Symptom) :-
    known(Symptom, no), !, fail.

ask(Symptom) :-
    write('Do you have '), write(Symptom), write('? (yes/no): '),
    read(Response),
    nl,
    asserta(known(Symptom, Response)),
    Response = yes.

disease(flu) :-
    ask(fever),
    ask(cough),
    ask(body_ache).

disease(covid19) :-
    ask(fever),
    ask(cough),
    ask(loss_of_taste),
    ask(breathing_difficulty).

disease(dengue) :-
    ask(high_fever),
    ask(joint_pain),
    ask(rash).

disease(malaria) :-
    ask(fever),
    ask(chills),
    ask(sweating).

disease(common_cold) :-
    ask(sneezing),
    ask(runny_nose),
    ask(mild_fever).

diagnose(Disease) :-
    disease(Disease), !.

diagnose(unknown).

start :-
    retractall(known(_, _)),
    write('--- DISEASE DIAGNOSIS EXPERT SYSTEM ---'), nl,
    diagnose(Disease),
    nl,
    explain(Disease).

explain(unknown) :-
    write('Could not determine the disease based on given symptoms.'), nl.

explain(Disease) :-
    write('You may have: '), write(Disease), nl.
