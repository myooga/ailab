#include <iostream>
#include <vector>
#include <queue>
#include <unordered_set>
#include <algorithm>
#include <cmath>
using namespace std;

struct State
{
    vector<vector<int>> board;
    int x, y;
};

vector<vector<int>> goal = {
    {1,2,3},
    {4,5,6},
    {7,8,0}
};

int dx[4] = {1, -1, 0, 0};
int dy[4] = {0, 0, 1, -1};

string board_to_string(vector<vector<int>> &b)
{
    string s = "";
    for (auto &row : b)
    {
        for (auto &val : row)
        {
            s += char(val + '0');
        }
    }
    return s;
}

int manhattan(vector<vector<int>> &b)
{
    int dist = 0;
    for (int i = 0; i < 3; i++)
    {
        for (int j = 0; j < 3; j++)
        {
            if (b[i][j] != 0)
            {
                int val = b[i][j];
                int tx = (val - 1) / 3;
                int ty = (val - 1) % 3;
                dist += abs(i - tx) + abs(j - ty);
            }
        }
    }
    return dist;
}

void print_board(vector<vector<int>> &b)
{
    cout << "\n";
    for (int i = 0; i < 3; i++)
    {
        for (int j = 0; j < 3; j++)
        {
            cout << b[i][j] << " ";
        }
        cout << "\n";
    }
}

void print_result(int goal_idx, vector<int> &parent, vector<State> &states)
{
    cout << "Goal Reached!\n";
    vector<int> path;
    int curr = goal_idx;
    while (curr != -1)
    {
        path.push_back(curr);
        curr = parent[curr];
    }
    reverse(path.begin(), path.end());
    for (int idx : path)
    {
        print_board(states[idx].board);
    }
}

void best_first_search(vector<vector<int>> start)
{
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    unordered_set<string> visited;
    vector<State> states;
    vector<int> parent;

    int sx, sy;
    for (int i = 0; i < 3; i++)
    {
        for (int j = 0; j < 3; j++)
        {
            if (start[i][j] == 0)
            {
                sx = i;
                sy = j;
            }
        }
    }

    states.push_back({start, sx, sy});
    parent.push_back(-1);
    pq.push({manhattan(start), 0});

    while (!pq.empty())
    {
        int idx = pq.top().second;
        pq.pop();

        vector<vector<int>> curr_board = states[idx].board;
        string key = board_to_string(curr_board);

        if (visited.count(key))
        {
            continue;
        }

        visited.insert(key);

        if (curr_board == goal)
        {
            print_result(idx, parent, states);
            return;
        }

        int x = states[idx].x;
        int y = states[idx].y;

        for (int i = 0; i < 4; i++)
        {
            int nx = x + dx[i];
            int ny = y + dy[i];

            if (nx >= 0 && ny >= 0 && nx < 3 && ny < 3)
            {
                vector<vector<int>> new_board = curr_board;
                swap(new_board[x][y], new_board[nx][ny]);

                string new_key = board_to_string(new_board);

                if (!visited.count(new_key))
                {
                    states.push_back({new_board, nx, ny});
                    parent.push_back(idx);

                    int new_idx = states.size() - 1;
                    pq.push({manhattan(new_board), new_idx});
                }
            }
        }
    }

    cout << "No solution found\n";
}
/*
T=O(b^d)
S=O(b^d)
*/
int main()
{
    vector<vector<int>> start = {
        {1,2,3},
        {4,0,6},
        {7,5,8}
    };

    best_first_search(start);

    return 0;
}
