#include <iostream>
#include <stack>
#include <set>
using namespace std;

struct State {
    int x, y, depth;
};

bool isGoal(State s, int d) {
    return (s.x == d || s.y == d);
}

void DLS(int cap1, int cap2, int d, int max_depth) {
    stack<State> st;
    set<pair<int, int>> visited;

    st.push({0, 0, 0});

    while (!st.empty()) {
        State curr = st.top();
        st.pop();

        if (visited.count({curr.x, curr.y})) {
            continue;
        }

        visited.insert({curr.x, curr.y});

        cout << "(" << curr.x << ", " << curr.y << ") at depth " << curr.depth << endl;

        if (isGoal(curr, d)) {
            cout << "Goal reached!" << endl;
            return;
        }

        if (curr.depth == max_depth) {
            continue;
        }

        st.push({cap1, curr.y, curr.depth + 1});
        st.push({curr.x, cap2, curr.depth + 1});
        st.push({0, curr.y, curr.depth + 1});
        st.push({curr.x, 0, curr.depth + 1});

        int pour1 = min(curr.x, cap2 - curr.y);
        st.push({curr.x - pour1, curr.y + pour1, curr.depth + 1});

        int pour2 = min(curr.y, cap1 - curr.x);
        st.push({curr.x + pour2, curr.y - pour2, curr.depth + 1});
    }

    cout << "No solution found within depth limit!" << endl;
}
/*Time complexity: O(b^d)
Space complexity: O(b·d)
*/
int main() {
    int cap1 = 4, cap2 = 3, d = 2;
    int max_depth = 3;

    DLS(cap1, cap2, d, max_depth);
    return 0;
}
