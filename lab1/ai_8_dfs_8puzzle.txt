#include <iostream>
#include <vector>
#include <stack>
#include <unordered_set>
#include <unordered_map>
#include <algorithm>
using namespace std;

struct State {
    vector<vector<int>> board;
    int x, y; 
};

vector<vector<int>> goal = {
    {1,2,3},
    {4,5,6},
    {7,8,0}
};

int dx[] = {-1, 1, 0, 0};
int dy[] = {0, 0, -1, 1};

string board_to_string(vector<vector<int>> &b) {
    string s = "";
    for (auto &row : b) {
        for (int val : row) {
            s += to_string(val);
        }
    }
    return s;
}

void print_path(unordered_map<string, string> &parent_state, string goal_state_str) {
    vector<string> path;

    while (goal_state_str != "") {
        path.push_back(goal_state_str);
        goal_state_str = parent_state[goal_state_str];
    }

    reverse(path.begin(), path.end());

    cout << "\nSolution Path:\n";
    for (string s : path) {
        for (int i = 0; i < 9; i++) {
            cout << s[i] << " ";
            if ((i + 1) % 3 == 0) cout << endl;
        }
        cout << "------\n";
    }
}

void dfs(State start) {
    stack<State> st;
    unordered_set<string> visited;
    unordered_map<string, string> parent_state;

    string start_state_str = board_to_string(start.board);
    parent_state[start_state_str] = "";

    st.push(start);

    while (!st.empty()) {
        State curr = st.top();
        st.pop();

        string curr_state_str = board_to_string(curr.board);

        if (visited.count(curr_state_str)) continue;
        visited.insert(curr_state_str);

        if (curr.board == goal) {
            cout << "Goal reached!\n";
            print_path(parent_state, curr_state_str);
            return;
        }

        for (int i = 0; i < 4; i++) {
            int nx = curr.x + dx[i];
            int ny = curr.y + dy[i];

            if (nx >= 0 && nx < 3 && ny >= 0 && ny < 3) {
                State next = curr;

                swap(next.board[curr.x][curr.y], next.board[nx][ny]);

                next.x = nx;
                next.y = ny;

                string next_state_str = board_to_string(next.board);

                if (!visited.count(next_state_str)) {
                    st.push(next);
                    parent_state[next_state_str] = curr_state_str;
                }
            }
        }
    }

    cout << "No solution found\n";
}
/*Time Complexity: O(b^d) ≈ O(9!)
Space Complexity: O(9!)*/
int main() {
    State start = {
        {{1,2,3},{4,5,0},{6,7,8}},
        1,2 
    };

    dfs(start);
    return 0;
}
