score_sentence([], 0, 0).

score_sentence([Word|Tail], PosScore, NegScore) :-
    positive_word(Word),
    score_sentence(Tail, TailPos, NegScore),
    PosScore is TailPos + 1.

score_sentence([Word|Tail], PosScore, NegScore) :-
    negative_word(Word),
    score_sentence(Tail, PosScore, TailNeg),
    NegScore is TailNeg + 1.

score_sentence([Word|Tail], PosScore, NegScore) :-
    \+ positive_word(Word),
    \+ negative_word(Word),
    score_sentence(Tail, PosScore, NegScore).

evaluate_sentiment(Pos, Neg, positive) :- Pos > Neg.
evaluate_sentiment(Pos, Neg, negative) :- Neg > Pos.
evaluate_sentiment(Pos, Neg, neutral)  :- Pos =:= Neg.

analyze(Sentence, Sentiment) :-
    score_sentence(Sentence, PosCount, NegCount),
    evaluate_sentiment(PosCount, NegCount, Sentiment).

check_sentiment(Sentence) :-
    analyze(Sentence, Sentiment),
    format('~n--- SENTIMENT ANALYSIS ---~n'),
    format('Input Phrase: ~w~n', [Sentence]),
    format('Calculated Sentiment: ~w~n', [Sentiment]).
