#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>

using namespace std;

bool win(const vector<int>& m_vec) {
    for (size_t i = 0; i < m_vec.size(); i++) {
        for (size_t j = i + 1; j < m_vec.size(); j++) {
            for (size_t k = j + 1; k < m_vec.size(); k++) {
                if (m_vec[i] + m_vec[j] + m_vec[k] == 15) return true;
            }
        }
    }
    return false;
}

void show(const char (&board)[9]) {
    cout << "\n";
    cout << " " << board[0] << " | " << board[1] << " | " << board[2] << "\n";
    cout << "---+---+---\n";
    cout << " " << board[3] << " | " << board[4] << " | " << board[5] << "\n";
    cout << "---+---+---\n";
    cout << " " << board[6] << " | " << board[7] << " | " << board[8] << "\n";
}

void play_non_ai() {
    int magic[9] = {8, 1, 6, 3, 5, 7, 4, 9, 2};
    char board[9] = {' ', ' ', ' ', ' ', ' ', ' ', ' ', ' ', ' '};
    vector<int> p1, p2;
    char p = 'X';
    int m;

    cout << "\n--- Magic Square Tic-Tac-Toe ---\n";
    cout << "Board positions are 0-8 (0=top-left, 8=bottom-right).\n";

    for (int t = 0; t < 9; t++) {
        show(board); 
        cout << "\nPlayer " << p << " move (0-8): ";
        cin >> m;

        if (m < 0 || m > 8 || board[m] != ' ') {
            cout << "Invalid move. Try again.\n";
            t--; 
            continue;
        }

        board[m] = p;

        if (p == 'X') {
            p1.push_back(magic[m]);
        } else {
            p2.push_back(magic[m]);
        }

        if (p1.size() >= 3 && win(p1)) {
            show(board);
            cout << "\nX wins!\n";
            return;
        }
        if (p2.size() >= 3 && win(p2)) {
            show(board);
            cout << "\nO wins!\n";
            return;
        }
        
        p = (p == 'X') ? 'O' : 'X';
    }
    
    show(board);
    cout << "\nIt's a draw!\n";
}

char ai_board[3][3];
const char HUMAN = 'X';
const char AI_PLAYER = 'O';

struct Move {
    int row, col;
};

void print_ai_board() {
    cout << "\n-------\n";
    for(int i = 0; i < 3; i++) {
        cout << "|" << ai_board[i][0] << "|" << ai_board[i][1] << "|" << ai_board[i][2] << "|\n";
        cout << "-------\n";
    }
}

char check_winner_ai() {
    for(int i = 0; i < 3; i++) {
        if(ai_board[i][0] != ' ' && ai_board[i][0] == ai_board[i][1] && ai_board[i][1] == ai_board[i][2]) return ai_board[i][0];
        if(ai_board[0][i] != ' ' && ai_board[0][i] == ai_board[1][i] && ai_board[1][i] == ai_board[2][i]) return ai_board[0][i];
    }
    if(ai_board[0][0] != ' ' && ai_board[0][0] == ai_board[1][1] && ai_board[1][1] == ai_board[2][2]) return ai_board[0][0];
    if(ai_board[0][2] != ' ' && ai_board[0][2] == ai_board[1][1] && ai_board[1][1] == ai_board[2][0]) return ai_board[0][2];
    
    bool is_draw = true;
    for(int i = 0; i < 3; i++) {
        for(int j = 0; j < 3; j++) {
            if(ai_board[i][j] == ' ') is_draw = false;
        }
    }
    if(is_draw) return 'D';
    
    return ' '; 
}

int minimax(bool is_max) {
    char winner = check_winner_ai();
    if(winner == AI_PLAYER) return 10;
    if(winner == HUMAN) return -10;
    if(winner == 'D') return 0;

    if(is_max) {
        int best = INT_MIN;
        for(int i = 0; i < 3; i++) {
            for(int j = 0; j < 3; j++) {
                if(ai_board[i][j] == ' ') {
                    ai_board[i][j] = AI_PLAYER;
                    best = max(best, minimax(false));
                    ai_board[i][j] = ' '; 
                }
            }
        }
        return best;
    } else {
        int best = INT_MAX;
        for(int i = 0; i < 3; i++) {
            for(int j = 0; j < 3; j++) {
                if(ai_board[i][j] == ' ') {
                    ai_board[i][j] = HUMAN;
                    best = min(best, minimax(true));
                    ai_board[i][j] = ' '; 
                }
            }
        }
        return best;
    }
}

Move find_best_move() {
    int best_val = INT_MIN;
    Move best_move = {-1, -1};

    for(int i = 0; i < 3; i++) {
        for(int j = 0; j < 3; j++) {
            if(ai_board[i][j] == ' ') {
                ai_board[i][j] = AI_PLAYER;
                int move_val = minimax(false);
                ai_board[i][j] = ' '; 
                
                if(move_val > best_val) {
                    best_move.row = i;
                    best_move.col = j;
                    best_val = move_val;
                }
            }
        }
    }
    return best_move;
}

void play_ai() {
    cout << "\n--- Minimax AI Tic-Tac-Toe (Human vs Computer) ---\n";
    
    for(int i = 0; i < 3; i++) {
        for(int j = 0; j < 3; j++) {
            ai_board[i][j] = ' ';
        }
    }

    while(true) {
        print_ai_board();
        cout << "Enter row and column (0-2) separated by a space: ";
        int row, col;
        
        if (!(cin >> row >> col)) {
            cin.clear();
            cin.ignore(10000, '\n');
            cout << "Invalid input. Returning to menu.\n";
            return;
        }
        
        if(row >= 0 && row < 3 && col >= 0 && col < 3 && ai_board[row][col] == ' ') {
            ai_board[row][col] = HUMAN;
        } else {
            cout << "Invalid move! Try again.\n";
            continue;
        }

        if(check_winner_ai() != ' ') break;

        cout << "AI is thinking...\n";
        Move best_move = find_best_move();
        if(best_move.row != -1 && best_move.col != -1) {
            ai_board[best_move.row][best_move.col] = AI_PLAYER;
        }
        
        if(check_winner_ai() != ' ') break;
    }

    print_ai_board();
    char winner = check_winner_ai();
    
    if(winner == HUMAN) {
        cout << "You Win!\n";
    } else if(winner == AI_PLAYER) {
        cout << "AI Wins!\n";
    } else {
        cout << "It's a Draw!\n";
    }
}

int main() {
    int choice;
    
    while(true) {
        cout << "1. Play Non-AI (Magic Square)\n";
        cout << "2. Play AI (Minimax Algorithm)\n";
        cout << "3. Exit\n";
        cout << "Enter your choice (1-3): ";
        
        if (!(cin >> choice)) {
            cin.clear();
            cin.ignore(10000, '\n'); 
            continue;
        }

        if (choice == 1) {
            play_non_ai();
        } else if (choice == 2) {
            play_ai();
        } else if (choice == 3) {
            cout << "Thanks for playing. Goodbye!\n";
            break;
        } else {
            cout << "Invalid choice. Please enter 1, 2, or 3.\n";
        }
    }

    return 0;
}
