#include <iostream>
#include <queue>
#include <vector>
#include <unordered_map>
#include <unordered_set>
#include <cmath>
#include <algorithm>
using namespace std;

struct Node
{
    string name;
    double h;

    bool operator>(const Node &other) const
    {
        return h > other.h;
    }
};

unordered_map<string, vector<string>> graph;
unordered_map<string, pair<int,int>> coord;

double heuristic(string current, string goal)
{
    int x1 = coord[current].first;
    int y1 = coord[current].second;

    int x2 = coord[goal].first;
    int y2 = coord[goal].second;

    return sqrt((x2 - x1)*(x2 - x1) + (y2 - y1)*(y2 - y1));
}

void best_first_search(string start, string goal)
{
    priority_queue<Node, vector<Node>, greater<Node>> pq;
    unordered_set<string> visited;
    unordered_map<string, string> parent;

    pq.push({start, heuristic(start, goal)});
    parent[start] = "";

    while (!pq.empty())
    {
        Node current = pq.top();
        pq.pop();

        string curr = current.name;

        if (visited.find(curr) != visited.end())
        {
            continue;
        }

        visited.insert(curr);

        if (curr == goal)
        {
            cout << "Goal reached\n";

            vector<string> path;
            string temp = goal;

            while (temp != "")
            {
                path.push_back(temp);
                temp = parent[temp];
            }

            reverse(path.begin(), path.end());

            cout << "Path: ";
            for (string node : path)
            {
                cout << node << " ";
            }
            cout << endl;

            return;
        }

        for (string next : graph[curr])
        {
            if (visited.find(next) == visited.end() && parent.find(next) == parent.end())
            {
                pq.push({next, heuristic(next, goal)});
                parent[next] = curr;
            }
        }
    }

    cout << "\nGoal not reachable\n";
}
/*
T=O(b^d), O(VlogV)
S=O((b^d)), O(V)
*/
int main()
{
    graph["A"] = {"B", "C"};
    graph["B"] = {"G"};
    graph["C"] = {"K"};
    graph["G"] = {"F"};
    graph["K"] = {};
    graph["F"] = {};

    coord["A"] = {0,0};
    coord["B"] = {2,2};
    coord["C"] = {1,4};
    coord["G"] = {5,5};
    coord["K"] = {2,6};
    coord["F"] = {7,7};

    best_first_search("A", "F");

    return 0;
}
