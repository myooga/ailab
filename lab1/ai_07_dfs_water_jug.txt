#include <iostream>
#include <stack>
#include <set>
using namespace std;

struct State {
    int x, y;
};

bool isGoal(State s, int d) {
    return (s.x == d || s.y == d);
}

void DFS(int cap1, int cap2, int d) {
    stack<State> st;
    set<pair<int, int>> visited;

    st.push({0, 0});

    while (!st.empty()) {
        State curr = st.top();
        st.pop();

        if (visited.count({curr.x, curr.y})) {
            continue;
        }

        visited.insert({curr.x, curr.y});

        cout << "(" << curr.x << ", " << curr.y << ")" << endl;

        if (isGoal(curr, d)) {
            cout << "Goal reached!" << endl;
            return;
        }

        st.push({cap1, curr.y});

        st.push({curr.x, cap2});

        st.push({0, curr.y});

        st.push({curr.x, 0});

        int pour1 = min(curr.x, cap2 - curr.y);
        st.push({curr.x - pour1, curr.y + pour1});

        int pour2 = min(curr.y, cap1 - curr.x);
        st.push({curr.x + pour2, curr.y - pour2});
    }

    cout << "No solution found!" << endl;
}
/*Time complexity: O(b^d)
Space complexity: O(b·d)
*/
int main() {
    int cap1 = 4, cap2 = 3, d = 2;
    DFS(cap1, cap2, d);
    return 0;
}
