#include <iostream>
#include <queue>
#include <map>
#include <vector>
#include <algorithm>
using namespace std;

struct State {
    vector<vector<int>> board;
    int x, y;
};

vector<vector<int>> goal = {
    {1,2,3},
    {4,5,6},
    {7,8,0}
};

bool isGoal(vector<vector<int>> &b) {
    return b == goal;
}

string serialize(vector<vector<int>> &b) {
    string s = "";
    for (auto &row : b) {
        for (auto &val : row) {
            s += to_string(val);
        }
    }
    return s;
}

vector<pair<State,string>> getNextStates(State curr) {
    vector<pair<State,string>> next;
    int dx[] = {-1,1,0,0};
    int dy[] = {0,0,-1,1};
    string moves[] = {"UP","DOWN","LEFT","RIGHT"};

    for (int i = 0; i < 4; i++) {
        int nx = curr.x + dx[i];
        int ny = curr.y + dy[i];

        if (nx >= 0 && nx < 3 && ny >= 0 && ny < 3) {
            State newState = curr;
            swap(newState.board[curr.x][curr.y], newState.board[nx][ny]);
            newState.x = nx;
            newState.y = ny;
            next.push_back({newState, moves[i]});
        }
    }
    return next;
}

void printBoard(vector<vector<int>> &b) {
    for (auto &row : b) {
        for (auto &val : row) {
            cout << val << " ";
        }
        cout << endl;
    }
    cout << "--------" << endl;
}

void BFS(vector<vector<int>> start) {
    queue<State> q;
    map<string, string> parent;
    map<string, string> moveTaken;

    State init;
    init.board = start;

    for (int i = 0; i < 3; i++) {
        for (int j = 0; j < 3; j++) {
            if (start[i][j] == 0) {
                init.x = i;
                init.y = j;
            }
        }
    }

    string startKey = serialize(start);
    q.push(init);
    parent[startKey] = "";

    while (!q.empty()) {
        State curr = q.front();
        q.pop();

        string currKey = serialize(curr.board);

        if (isGoal(curr.board)) {
            cout << "Goal reached!\n\n";

            vector<string> path;
            string temp = currKey;

            while (temp != "") {
                path.push_back(temp);
                temp = parent[temp];
            }

            reverse(path.begin(), path.end());

            for (int i = 0; i < path.size(); i++) {
                string s = path[i];
                vector<vector<int>> board(3, vector<int>(3));
                int k = 0;

                for (int r = 0; r < 3; r++) {
                    for (int c = 0; c < 3; c++) {
                        board[r][c] = s[k++] - '0';
                    }
                }

                cout << "Step " << i << endl;
                printBoard(board);

                if (i > 0) {
                    cout << "Move: " << moveTaken[s] << endl;
                }
            }

            cout << "Total moves: " << path.size() - 1 << endl;
            return;
        }

        auto neighbors = getNextStates(curr);

        for (auto &p : neighbors) {
            State nextState = p.first;
            string move = p.second;

            string key = serialize(nextState.board);

            if (parent.find(key) == parent.end()) {
                q.push(nextState);
                parent[key] = currKey;
                moveTaken[key] = move;
            }
        }
    }

    cout << "No solution found!" << endl;
}
/*
T=O(B^D),O(9!),O(V+E)
S=O(B^D),O(9!)
*/
int main() {
    vector<vector<int>> start = {
        {1,2,3},
        {4,0,5},
        {7,8,6}
    };
    BFS(start);
    return 0;
}
