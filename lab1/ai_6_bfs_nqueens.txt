#include <iostream>
#include <vector>
#include <queue>
using namespace std;

#define N 8

bool isSafe(vector<int> &state, int row, int col) {
    for (int i = 0; i < row; i++) {
        int prev_col = state[i];

        if (prev_col == col) {
            return false;
        }

        if (abs(prev_col - col) == abs(i - row)) {
            return false;
        }
    }
    return true;
}

void printSolution(vector<int> &state) {
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            if (state[i] == j) {
                cout << "Q ";
            } else {
                cout << ". ";
            }
        }
        cout << endl;
    }
    cout << "------------------" << endl;
}

void solveNQueensBFS() {
    queue<vector<int>> q;

    q.push(vector<int>());

    while (!q.empty()) {
        vector<int> curr = q.front();
        q.pop();

        int row = curr.size();

        if (row == N) {
            printSolution(curr);
            continue;
        }

        for (int col = 0; col < N; col++) {
            if (isSafe(curr, row, col)) {
                vector<int> next = curr;
                next.push_back(col);
                q.push(next);
            }
        }
    }
}

int main() {
    solveNQueensBFS();
    return 0;
}
