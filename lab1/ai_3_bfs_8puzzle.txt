#include <iostream>
#include <queue>
#include <set>
#include <vector>
using namespace std;

struct State {
    vector<vector<int>> board;
    int x, y; // position of zero
    int level;
};

vector<vector<int>> goal = {
    {1,2,3},
    {4,5,6},
    {7,8,0}
};

bool isGoal(vector<vector<int>> &b) {
    return b == goal;
}

string serialize(vector<vector<int>> &b) {
    string s = "";
    for (auto &row : b) {
        for (auto &val : row) {
            s += to_string(val);
        }
    }
    return s;
}

vector<State> getNextStates(State curr) {
    vector<State> next;
    int dx[] = {-1,1,0,0};
    int dy[] = {0,0,-1,1};

    for (int i = 0; i < 4; i++) {
        int nx = curr.x + dx[i];
        int ny = curr.y + dy[i];

        if (nx >= 0 && nx < 3 && ny >= 0 && ny < 3) {
            State newState = curr;
            swap(newState.board[curr.x][curr.y], newState.board[nx][ny]);
            newState.x = nx;
            newState.y = ny;
            newState.level = curr.level + 1;
            next.push_back(newState);
        }
    }
    return next;
}

void BFS(vector<vector<int>> start) {
    queue<State> q;
    set<string> visited;

    State init;
    init.board = start;
    init.level = 0;

    // find zero position
    for (int i = 0; i < 3; i++) {
        for (int j = 0; j < 3; j++) {
            if (start[i][j] == 0) {
                init.x = i;
                init.y = j;
            }
        }
    }

    q.push(init);
    visited.insert(serialize(start));

    while (!q.empty()) {
        State curr = q.front();
        q.pop();

        if (isGoal(curr.board)) {
            cout << "Goal reached at level: " << curr.level << endl;
            return;
        }

        vector<State> neighbors = getNextStates(curr);

        for (auto &n : neighbors) {
            string key = serialize(n.board);
            if (visited.find(key) == visited.end()) {
                q.push(n);
                visited.insert(key);
            }
        }
    }

    cout << "No solution found!" << endl;
}

int main() {
    vector<vector<int>> start = {
        {1,2,3},
        {4,0,5},
        {7,8,6}
    };

    BFS(start);

    return 0;
}
