% Facts: Birds
bird(sparrow).
bird(pigeon).
bird(parrot).
bird(penguin).
bird(ostrich).

% Properties
cannot_fly(penguin).
cannot_fly(ostrich).

can_swim(penguin).

% Rule: A bird can fly if it is not marked as cannot_fly
can_fly(X) :- bird(X), \+ cannot_fly(X).

% Classification Rules
type(X, flying_bird) :- can_fly(X).
type(X, flightless_bird) :- bird(X), cannot_fly(X).
type(X, swimming_bird) :- bird(X), can_swim(X).
