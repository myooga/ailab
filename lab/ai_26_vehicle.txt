% --------- Vehicle Classification System ---------

% Facts: Vehicles
vehicle(car).
vehicle(bike).
vehicle(truck).
vehicle(bus).
vehicle(bicycle).
vehicle(scooter).
vehicle(train).

% Properties
wheels(car, 4).
wheels(bike, 2).
wheels(truck, 6).
wheels(bus, 6).
wheels(bicycle, 2).
wheels(scooter, 2).
wheels(train, many).

engine(car).
engine(bike).
engine(truck).
engine(bus).
engine(scooter).
engine(train).

no_engine(bicycle).

public_transport(bus).
public_transport(train).

heavy_vehicle(truck).
heavy_vehicle(bus).
heavy_vehicle(train).

% --------- Rules ---------

motorized(X) :- engine(X).

non_motorized(X) :- no_engine(X).

light_vehicle(X) :- vehicle(X), \+ heavy_vehicle(X).

type(X, motor_vehicle) :- motorized(X).
type(X, non_motor_vehicle) :- non_motorized(X).
type(X, heavy_vehicle) :- heavy_vehicle(X).
type(X, light_vehicle) :- light_vehicle(X).
type(X, public_transport) :- public_transport(X).
