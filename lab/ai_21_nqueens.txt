#include <iostream>
#include <cmath>
#include <vector>
using namespace std;
int n;
vector<int> x;

bool Place(int k) {
    for (int j = 0; j < k; j++) {
        if (x[j] == x[k] || abs(x[j] - x[k]) == abs(j - k)) {
            return false;
        }
    }
    return true;
}

void Backtrack(int k) {
    for (int col = 0; col < n; col++) {
        
        x[k] = col;
        if (Place(k)) {

            if (k == n - 1) {
                for (int i = 0; i < n; i++) {
                    cout << x[i] << " ";
                }
                cout << endl;
            }
            else {
                Backtrack(k + 1);
            }
        }
    }
}

int main() {
    cout << "Enter number of queens (n): ";
    cin >> n;
    x.resize(n);
    Backtrack(0);
    return 0;
}
