#include <iostream>
#include <queue>
#include <set>
#include <vector>

using namespace std;

struct State {
    int x, y;
    vector<pair<int, int>> path;
};

void solve(int cap1, int cap2, int target) {
    queue<State> q;
    set<pair<int, int>> visited;

    q.push({0, 0, {{0, 0}}});
    visited.insert({0, 0});

    while (!q.empty()) {
        State curr = q.front();
        q.pop();

        if (curr.x == target || curr.y == target) {
            for (int i = 0; i < curr.path.size(); i++) {
                cout << "(" << curr.path[i].first << ", " << curr.path[i].second << ")" << endl;
            }
            return;
        }

        int next_states[6][2] = {
            {cap1, curr.y}, 
            {curr.x, cap2}, 
            {0, curr.y},    
            {curr.x, 0},    
            {curr.x + min(curr.y, cap1 - curr.x), curr.y - min(curr.y, cap1 - curr.x)},
            {curr.x - min(curr.x, cap2 - curr.y), curr.y + min(curr.x, cap2 - curr.y)}
        };

        for (int i = 0; i < 6; i++) {
            int nx = next_states[i][0];
            int ny = next_states[i][1];

            if (visited.find({nx, ny}) == visited.end()) {
                visited.insert({nx, ny});
                vector<pair<int, int>> next_path = curr.path;
                next_path.push_back({nx, ny});
                q.push({nx, ny, next_path});
            }
        }
    }
    cout << "No solution" << endl;
}

int main() {
    int jug1 = 4, jug2 = 3, target = 2;
    solve(jug1, jug2, target);
    return 0;
}
