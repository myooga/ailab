#include <iostream>
#include <vector>
using namespace std;

int n, m;
vector<vector<int>> g;
vector<int> x;

bool Place(int k) {
    for (int j = 0; j < k; j++) {
        if (g[k][j] && x[j] == x[k]) return false;
    }
    return true;
}

void Backtrack(int k) {
    for (int c = 0; c < m; c++) {
        x[k] = c;
        if (Place(k)) {
            if (k == n - 1) {
                for (int i = 0; i < n; i++) {
                    cout << x[i] << " ";
                }
                cout << "\n";
            }
            else Backtrack(k + 1);
        }
    }
}

int main() {
    cin >> n >> m;

    g.assign(n, vector<int>(n, 0));
    x.assign(n, 0);

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> g[i][j];
        }
    }

    Backtrack(0);
    return 0;
}
